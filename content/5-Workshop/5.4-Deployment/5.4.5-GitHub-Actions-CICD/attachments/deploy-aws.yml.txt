name: CI/CD Pipeline — Couple Assistant AWS EC2 Deployment

on:
  push:
    branches: [ main, feature/unified-itinerary-ux ]
  pull_request:
    branches: [ main ]
  workflow_dispatch:
    inputs:
      deploy_production:
        description: 'Deploy to AWS Production (Singapore ap-southeast-1)'
        required: false
        type: boolean
        default: true

permissions:
  id-token: write   # Required for AWS OIDC authentication
  contents: read    # Required for actions/checkout

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  # ==============================================================================
  # STEP 1 — CONTINUOUS INTEGRATION (CI)
  # ==============================================================================
  ci-quality-and-build:
    name: "CI — Tests, Security & Docker Build"
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Set up Python 3.11
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: "pip"

      - name: Install Backend Test Dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r backend/requirements.txt pytest

      - name: Run Backend Automated Tests
        working-directory: backend
        run: |
          python -m pytest tests/ -v

      - name: "Security Scan — Secret & Dependency Check"
        run: |
          echo "Checking for committed secrets or private keys..."
          if git log -n 5 --oneline | grep -iE "(secret|password|credential)" > /dev/null; then
            echo "::warning::Check recent commit messages for sensitive wording."
          fi
          # Ensure no database files committed
          if git ls-files | grep -E "\.(db|sqlite|sqlite3|db\.bak)$"; then
            echo "Error: Database files found tracked in Git!"
            exit 1
          fi

      - name: Set up Node.js 20
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: "npm"
          cache-dependency-path: frontend/package-lock.json

      - name: "Install Frontend Dependencies & Build SPA"
        working-directory: frontend
        run: |
          npm ci || npm install
          npm run build

      - name: Upload Frontend Artifact
        uses: actions/upload-artifact@v4
        with:
          name: frontend-dist
          path: frontend/dist/
          retention-days: 3

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Build Backend Docker Image (Validation)
        uses: docker/build-push-action@v5
        with:
          context: ./backend
          push: false
          tags: couplelife-backend:test-ci

      - name: Build AI Service Docker Image (Validation)
        uses: docker/build-push-action@v5
        with:
          context: ./ai-service
          push: false
          tags: couplelife-ai-service:test-ci

  # ==============================================================================
  # STEP 2 — CONTINUOUS DEPLOYMENT (CD) — AWS SINGAPORE
  # ==============================================================================
  cd-deploy-aws:
    name: CD — Deploy to AWS EC2 (Singapore)
    needs: ci-quality-and-build
    runs-on: ubuntu-latest
    # Automatically triggers on push to main OR manual trigger with deploy_production=true
    if: github.ref == 'refs/heads/main' && (github.event_name == 'push' || (github.event_name == 'workflow_dispatch' && inputs.deploy_production != false))
    # Production environment requires manual review & approval gate in GitHub Settings
    environment: production
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Download Frontend Build Artifact
        uses: actions/download-artifact@v4
        with:
          name: frontend-dist
          path: frontend/dist

      - name: Configure AWS Credentials (OIDC Authentication)
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::762948750428:role/CoupleLifeAI-GitHubActions-DeployRole
          role-session-name: GitHubActions-CoupleLife-Deploy
          aws-region: ap-southeast-1

      - name: Log in to Amazon ECR
        id: login-ecr
        uses: aws-actions/amazon-ecr-login@v2

      - name: Build and Push Backend Docker Image to ECR
        env:
          ECR_REGISTRY: ${{ steps.login-ecr.outputs.registry }}
          ECR_REPOSITORY: couplelife-backend
          IMAGE_TAG: ${{ github.sha }}
        run: |
          # Use SHA commit tag only — ECR repository is IMMUTABLE (no 'latest' tag)
          if aws ecr describe-images --repository-name $ECR_REPOSITORY --image-ids imageTag=$IMAGE_TAG >/dev/null 2>&1; then
            echo "Image $IMAGE_TAG already exists in $ECR_REPOSITORY, skipping build and push."
          else
            docker build -t $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG ./backend
            docker push $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG
          fi

      - name: Build and Push AI Service Docker Image to ECR
        env:
          ECR_REGISTRY: ${{ steps.login-ecr.outputs.registry }}
          ECR_REPOSITORY: couplelife-ai-service
          IMAGE_TAG: ${{ github.sha }}
        run: |
          # Use SHA commit tag only — ECR repository is IMMUTABLE (no 'latest' tag)
          if aws ecr describe-images --repository-name $ECR_REPOSITORY --image-ids imageTag=$IMAGE_TAG >/dev/null 2>&1; then
            echo "Image $IMAGE_TAG already exists in $ECR_REPOSITORY, skipping build and push."
          else
            docker build -t $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG ./ai-service
            docker push $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG
          fi

      - name: "Deploy Frontend to S3 & Invalidate CloudFront"
        env:
          S3_BUCKET: couplelife-frontend-762948750428
          CLOUDFRONT_ID: E1WLIFW7QBA3SY
        run: |
          echo "Syncing frontend assets to private S3 bucket..."
          aws s3 sync frontend/dist/ s3://$S3_BUCKET/ --delete --cache-control "public, max-age=31536000, immutable"
          # Ensure index.html is never cached permanently
          aws s3 cp frontend/dist/index.html s3://$S3_BUCKET/index.html --cache-control "no-cache, no-store, must-revalidate"
          
          echo "Creating CloudFront cache invalidation..."
          if [ -n "$CLOUDFRONT_ID" ]; then aws cloudfront create-invalidation --distribution-id "$CLOUDFRONT_ID" --paths "/*" || true; fi

      - name: Execute Database Migrations (One-time Pre-Release via SSM)
        env:
          ASG_NAME: CoupleLifeAI-ASG
          ECR_REGISTRY: ${{ steps.login-ecr.outputs.registry }}
          IMAGE_TAG: ${{ github.sha }}
        run: |
          echo "Finding one online instance from ASG for migration..."
          TARGET_INSTANCE=$(aws autoscaling describe-auto-scaling-groups \
            --auto-scaling-group-names "$ASG_NAME" \
            --query "AutoScalingGroups[0].Instances[?LifecycleState=='InService'].InstanceId | [0]" \
            --output text)
          
          if [ -z "$TARGET_INSTANCE" ] || [ "$TARGET_INSTANCE" == "None" ]; then
            echo "Error: No InService instance found in $ASG_NAME"
            exit 1
          fi

          echo "Executing database migration on $TARGET_INSTANCE..."
          COMMAND_ID=$(aws ssm send-command \
            --instance-ids "$TARGET_INSTANCE" \
            --document-name "AWS-RunShellScript" \
            --parameters "commands=[
              \"aws ecr get-login-password --region ap-southeast-1 | docker login --username AWS --password-stdin $ECR_REGISTRY\",
              \"docker pull $ECR_REGISTRY/couplelife-backend:$IMAGE_TAG\",
              \"docker run --rm --env-file /etc/couplelife/backend.env $ECR_REGISTRY/couplelife-backend:$IMAGE_TAG alembic upgrade head\"
            ]" \
            --query "Command.CommandId" --output text)
          
          aws ssm wait command-executed --command-id "$COMMAND_ID" --instance-id "$TARGET_INSTANCE"
          echo "Database migration completed successfully."

      - name: Synchronize Launch Template & SSM Parameter for Scale-Out
        env:
          IMAGE_TAG: ${{ github.sha }}
          LT_NAME: CoupleLifeAI-LT
        run: |
          echo "Updating SSM parameter /couplelife/production/image-tag with new commit SHA..."
          aws ssm put-parameter \
            --name "/couplelife/production/image-tag" \
            --value "$IMAGE_TAG" \
            --type "String" \
            --overwrite || true

          echo "Tagging Launch Template $LT_NAME with latest commit..."
          LT_ID=$(aws ec2 describe-launch-templates --launch-template-names "$LT_NAME" --query "LaunchTemplates[0].LaunchTemplateId" --output text 2>/dev/null || echo "None")
          if [ "$LT_ID" != "None" ] && [ -n "$LT_ID" ]; then
            aws ec2 create-tags --resources "$LT_ID" --tags "Key=LatestDeployedCommit,Value=$IMAGE_TAG" || true
          fi

      - name: Deploy Docker Containers to ASG Instances (Zero-Downtime Rolling Update & Rollback)
        env:
          ASG_NAME: CoupleLifeAI-ASG
          TG_ARN: arn:aws:elasticloadbalancing:ap-southeast-1:762948750428:targetgroup/CoupleLifeAI-TG/ef9263a80cfaef5e
          ECR_REGISTRY: ${{ steps.login-ecr.outputs.registry }}
          IMAGE_TAG: ${{ github.sha }}
        run: |
          echo "Retrieving previous stable image tag for rollback protection..."
          PREV_TAG=$(aws ssm get-parameter --name "/couplelife/production/image-tag" --query "Parameter.Value" --output text 2>/dev/null || echo "f4ab56c")
          echo "Previous stable tag: $PREV_TAG. Deploying target commit tag: $IMAGE_TAG"

          INSTANCES=$(aws autoscaling describe-auto-scaling-groups \
            --auto-scaling-group-names "$ASG_NAME" \
            --query "AutoScalingGroups[0].Instances[?LifecycleState=='InService'].InstanceId" \
            --output text)
          
          if [ -z "$INSTANCES" ] || [ "$INSTANCES" == "None" ]; then
            echo "Error: No InService instances found in $ASG_NAME!"
            exit 1
          fi

          INSTANCE_COUNT=$(echo "$INSTANCES" | wc -w)
          echo "Found $INSTANCE_COUNT instances: $INSTANCES. Beginning zero-downtime rolling deployment..."

          DEPLOY_B64=$(base64 -w 0 deploy/aws/scripts/rolling-deploy-node.sh)
          ROLLBACK_B64=$(base64 -w 0 deploy/aws/scripts/rolling-rollback-node.sh)

          CURRENT_INDEX=1
          for INSTANCE_ID in $INSTANCES; do
            echo "=========================================================="
            echo "[Node $CURRENT_INDEX / $INSTANCE_COUNT] Deploying $IMAGE_TAG to $INSTANCE_ID..."
            echo "=========================================================="
            
            CMD_ID=$(aws ssm send-command \
              --instance-ids "$INSTANCE_ID" \
              --document-name "AWS-RunShellScript" \
              --parameters "commands=[\"echo $DEPLOY_B64 | base64 -d | bash -s -- $IMAGE_TAG $PREV_TAG\"]" \
              --query "Command.CommandId" --output text)
            
            echo "Waiting for node deployment execution (SSM Command: $CMD_ID)..."
            aws ssm wait command-executed --command-id "$CMD_ID" --instance-id "$INSTANCE_ID"
            
            CMD_STATUS=$(aws ssm get-command-invocation --command-id "$CMD_ID" --instance-id "$INSTANCE_ID" --query "Status" --output text 2>/dev/null || echo "Unknown")
            echo "Node deployment command finished with status: $CMD_STATUS"
            if [ "$CMD_STATUS" != "Success" ]; then
              echo "Error: Deployment command on $INSTANCE_ID failed with status $CMD_STATUS!"
              aws ssm get-command-invocation --command-id "$CMD_ID" --instance-id "$INSTANCE_ID" --query "StandardErrorContent" --output text || true
              exit 1
            fi
            
            echo "Verifying ALB Target Group health status for node $INSTANCE_ID..."
            IS_HEALTHY=0
            for attempt in $(seq 1 12); do
              sleep 10
              STATE=$(aws elbv2 describe-target-health --target-group-arn "$TG_ARN" --targets Id="$INSTANCE_ID" --query "TargetHealthDescriptions[0].TargetHealth.State" --output text 2>/dev/null || echo "unknown")
              echo "Attempt $attempt: Instance $INSTANCE_ID status = $STATE"
              if [ "$STATE" == "healthy" ]; then
                IS_HEALTHY=1
                echo "Instance $INSTANCE_ID is confirmed HEALTHY in ALB Target Group!"
                break
              fi
            done

            if [ "$IS_HEALTHY" -ne 1 ]; then
              echo "CRITICAL: Instance $INSTANCE_ID failed health check! Initiating immediate ROLLBACK to $PREV_TAG..."
              aws ssm send-command \
                --instance-ids "$INSTANCE_ID" \
                --document-name "AWS-RunShellScript" \
                --parameters "commands=[\"echo $ROLLBACK_B64 | base64 -d | bash -s -- $PREV_TAG\"]"
              echo "Rollback command executed on $INSTANCE_ID. Aborting pipeline before affecting other nodes."
              exit 1
            fi

            CURRENT_INDEX=$((CURRENT_INDEX + 1))
          done
          echo "Rolling deployment successfully completed across all $INSTANCE_COUNT nodes with ALB Healthy confirmation."

      - name: Post-Deployment Resilient Health Check
        env:
          HEALTH_URL: https://d2e5dd7odfq09h.cloudfront.net/api/v1/health
        run: |
          echo "Verifying health check at $HEALTH_URL with retry mechanism..."
          MAX_RETRIES=6
          RETRY_DELAY=10
          SUCCESS=0

          for i in $(seq 1 $MAX_RETRIES); do
            echo "Health check attempt $i of $MAX_RETRIES..."
            HTTP_STATUS=$(curl -s -o /dev/null -w "%{http_code}" "$HEALTH_URL" || echo "000")
            if [ "$HTTP_STATUS" -eq 200 ]; then
              echo "Health check PASSED with HTTP 200 on attempt $i!"
              SUCCESS=1
              break
            else
              echo "Health check returned HTTP $HTTP_STATUS. Retrying in ${RETRY_DELAY}s..."
              sleep $RETRY_DELAY
            fi
          done

          if [ "$SUCCESS" -ne 1 ]; then
            echo "Error: Health check FAILED after $MAX_RETRIES attempts (Last HTTP: $HTTP_STATUS)."
            exit 1
          fi
          echo "Deployment verified live and healthy!"
